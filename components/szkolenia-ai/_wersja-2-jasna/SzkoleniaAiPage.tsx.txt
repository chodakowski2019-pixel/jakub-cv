// Szkolenie AI dla firm. Wersja 2 wyglądu (21.11.2026, USER_001: „bardzo nowoczesny, przyjazny dla oka,

// Poprzednia wersja (żółty „ROZKŁAD”) leży w _wersja-rozklad/.
// Kolejność: hero → dlaczego teraz → co zostaje → rozkład dnia → przebieg → prowadzący → dla kogo →

// Twarda spacja: jednoliterowe słowa i liczby nie łamią się między wierszami.
const TS = String.fromCharCode(160);
const s = (t: string) => t.replace(/(^|\s)([wzioauWZIOAU])\s/g, `$1$2${TS}`).replace(/(\d)\s(\d{3})(?!\d)/g, `$1${TS}$2`).replace(/(\d)\s(r\.)/g, `$1${TS}$2`);

const Strzalka = () => (
  <svg className="strz" viewBox="0 0 16 16" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
    <path d="M3 8h10M9 4l4 4-4 4" />
  </svg>
);
const Ptaszek = () => (
  <svg className="znak tak" viewBox="0 0 20 20" aria-hidden="true">
    <circle cx="10" cy="10" r="10" />
    <path d="M5.8 10.4l2.7 2.7 5.7-6" fill="none" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round" />
  </svg>
);
const Krzyzyk = () => (
  <svg className="znak nie" viewBox="0 0 20 20" aria-hidden="true">
    <circle cx="10" cy="10" r="10" />
    <path d="M7 7l6 6M13 7l-6 6" fill="none" strokeWidth="1.8" strokeLinecap="round" />
  </svg>
);

function Glowa({ nad, tytul, id, lead }: { nad: string; tytul: string; id: string; lead?: string }) {
  return (
    <div className="sek-glowa poj">
      <p className="nad">{nad}</p>
      <h2 id={id}>{s(tytul)}</h2>
      {lead && <p className="lead">{s(lead)}</p>}
    </div>
  );
}
