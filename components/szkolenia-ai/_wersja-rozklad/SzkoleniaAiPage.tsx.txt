// Jednoliterowe słowa nie zostają na końcu wiersza.
const TS = String.fromCharCode(160); // twarda spacja
const s = (t: string) => t.replace(/(^|\s)([wzioauWZIOAU])\s/g, `$1$2${TS}`).replace(/(\d)\s(\d{3})(?!\d)/g, `$1${TS}$2`).replace(/(\d)\s(r\.)/g, `$1${TS}$2`);

    <path d="M2 8h12M9 3l5 5-5 5" />
  </svg>
);

function Glowa({ nr, tytul, id }: { nr: string; tytul: string; id: string }) {
  return (
    <div className="sek-glowa poj">
      <span className="nr" aria-hidden="true">{nr}</span>
      <h2 id={id}>{s(tytul)}</h2>
    </div>
  );
}

function Liczby({ lista }: { lista: { n: string; opis: string; zrodlo: string }[] }) {
  return (
    <div className="liczby">
      {lista.map((l) => (
        <div className="liczba poj" key={l.n}>
          <b>{l.n}</b>
          <p>{s(l.opis)}</p>
          <small>Źródło: {l.zrodlo}</small>
        </div>
      ))}
    </div>
  );
}
